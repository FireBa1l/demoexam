        <Button Click="DeleteButton_Click" Width="60" Grid.Column="2" Content="Удалить" HorizontalAlignment="Center" VerticalAlignment="Top" Margin="0,25,0,0" Grid.Row="1"/>

using System;
using System.Data;
using System.Data.SqlClient;
using System.Windows;
using System.Windows.Controls;

namespace Umey
{
    public partial class TableWinowAdm : Window
    {
        SqlDataAdapter adapter;
        DataTable table;
        public TableWinowAdm()
        {
            InitializeComponent();
            string con = "Data Source=DESKTOP-6R8LE85;Initial Catalog=tachki;Integrated Security=True";
            string query = "SELECT * FROM Cars";
            adapter = new SqlDataAdapter(query, con);
            table = new DataTable();
            SqlCommandBuilder b = new SqlCommandBuilder(adapter);
            adapter.Fill(table);
            dg.ItemsSource = table.DefaultView;
            dg.CanUserAddRows = false;
        }
        private void save_click(object sender, RoutedEventArgs e)
        {
            try
            {
                adapter.Update(table);
                MessageBox.Show("Сохранено!");
            }
            catch (Exception ex)
            {
                MessageBox.Show("Ошибка: " + ex.Message);
            }
        }
        private void AddButton_click(object sender, RoutedEventArgs e)
        {
            DataRow newRow = table.NewRow();
            table.Rows.Add(newRow);
        }
        private void DeleteButton_Click(object sender, RoutedEventArgs e)
        {
            if (dg.SelectedItem != null)
                ((DataRowView)dg.SelectedItem).Row.Delete();
        }
        private void searchBox_TextChanged(object sender, TextChangedEventArgs e)
        {
            if (table == null) return;

            if (string.IsNullOrEmpty(searchBox.Text))
                table.DefaultView.RowFilter = "";
            else
                table.DefaultView.RowFilter =
                    $"Name LIKE '%{searchBox.Text}%' OR " +
                    $"Number LIKE '%{searchBox.Text}%' OR " +
                    $"Color LIKE '%{searchBox.Text}%'";
        }
    }
}
